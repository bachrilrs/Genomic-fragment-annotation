Method name ( ): extract_infos_GenemarkHMM
extract_infos_GenemarkHMM(input_file ↓ )

Description:
Extracts the sequence identifier, program name, and CDS information
    from a genemarkhmm output file.

Header:
input_file ↓ : path to the genemarkhmm output file
Return value: Tuple containing:
    sequence identifier (seqid)
    program name (source)
    list of CDS (cds_list)

Body
Local variables

- seqid: String
- source: String
- cds_list: List
- line: String
- words: List
- fields: List

Instructions
seqid <- empty string
source <- empty string
cds_list <- empty list



OPEN input_file FOR READING DO
    state <- 'outside'
    FOR EACH line IN file DO
        line <- stripSpacesAtStartAndEnd(line)
        IF line contains  'version:' THEN
          words <- Split(line, " ")
          source <- words[0]
        END IF
        IF line starts with 'FASTA definition line:' THEN

            seq_match<- ExtractWithRegex(':\s+(\S+)', line)
        
            IF seq_match returns a value THEN
                seqid <- extracted value
            END IF
        END IF
        IF line starts with 'Predicted genes:' THEN
            state <- 'data'
            CONTINUE
        END IF

        IF state = 'data' AND FirstLetter(line) is a digit THEN
            fields <- SplitBySpaces(line)
            ADD fields to cds_list
        END IF
    END FOR
CLOSE file
Return (seqid, source, cds_list)










Method name ( ): taille_seq
taille_seq(seq_fasta_path ↓ )
Description:
Calculates the sequence length from a FASTA file.

Header:
seq_fasta_path ↓ : path to the FASTA file
Return value: Integer representing the sequence length

Body
Local variables
- taille: Integer
- line: String


Instructions
taille <- 0
OPEN seq_fasta_path FOR READING Do
    FOR EACH line IN file DO
        line <- stripSpacesAtStartAndEnd(line)
        IF line does not start with '>' THEN
            taille <- taille + Length(line)
        END IF
    END FOR
CLOSE file
Return taille








Method name ( ): Nettoyer_int
Nettoyer_int(Character)
Description:
Removes the '>' and '<' characters from a string before converting it to an integer.

Header:
Character ↓ : String representing a number potentially containing '>' or '<'
Return value: Integer
Body
Local variables
- nombre_str : String
- nombre_int : Integer
Instructions
nombre_str <- Replace(Character, '>', '')
nombre_str <- Replace(nombre_str, '<', '')
nombre_int <- Integer(nombre_str)
Return nombre_int









Method name ( ): Write_GFF3_GeneMark
Write_GFF3_GeneMark(input_file ↓ , output_file ↓ , seq_fasta (Undefined by default) ↓ )
Description:
// Writes the GFF3 annotations of CDS extracted from a GeneMark.hmm output file

// This function uses extract_infos_GenemarkHMM and taille_seq to obtain the necessary information.

Header:
input_file ↓ : path to the GeneMark.hmm output file
output_file ↓ : path to the GFF3 output file
Returns success message

Body
Local variables
- seqid : String
- source : String
- cds_list : List
- taille : Integer
- i : Integer
- fields : List
- left : Integer
- right : Integer
- start : Integer
- end : Integer
- score : String
- strand : String
- phase : Integer
- classe : String
- atributes : String
- ecrire_ligne : String

Instructions

seqid, source, cds_list <- extract_infos_GenemarkHMM(input_file)
Taille <- Undefined

IF seq_fasta exists THEN
    taille <- taille_seq(seq_fasta)
END IF

OPEN output_file FOR WRITING do

  WRITE "##gff-version 3\n" TO output_file

  IF taille exists THEN
      WRITE '##sequence-region ' + seqid + ' 1 ' + taille + '\n' TO output_file
  END IF

  FOR i from 0 to length(cds_list) − 1 DO fields <- cds_list[i]
      left <- Nettoyer_int(fields[0])
      right <- Nettoyer_int(fields[1])
      start <- Minimum(left, right)
      end <- Maximum(left, right)


      score <- '.'
      strand <- fields[1]

      IF strand = '+' THEN
         phase <- (start - 1) MOD 3
        ELSE
            phase <- (end - 1) MOD 3
      END IF

      IF length(cds_list) >= 6 THEN
          classe <- fields[5]
        ELSE
          classe <- '.'
      END IF
    
      atributes <- 'ID=cds' + (i + 1) + ';Note=' + source + '_prediction;Class=' + classe
      
      ecrire_ligne <- seqid + '\t' + source + '\t' + 'CDS' + '\t' + start + '\t' + end + '\t' + score + '\t' + strand + '\t' + phase + '\t' + atributes + '\n'
      WRITE ecrire_ligne TO output_file
  END FOR
CLOSE output_file
Return 'GFF3 file ' + output_file + ' written successfully'
End Instructions
