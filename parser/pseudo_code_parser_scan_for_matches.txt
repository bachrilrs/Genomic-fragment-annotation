Import re (Regular expressions)



Method name ( ): taille_seq
taille_seq(seq_fasta_file)
Description :
Calculates the sequence length from a FASTA file.

Header:
seq_fasta_file ↓ : path to the FASTA file
Return value: Integer representing the sequence length

Body
Local variables
- taille: Integer
- line: String


Instructions
taille <- 0
OPEN seq_fasta_file FOR READING DO
    FOR EACH line IN file DO
        line <- stripSpacesAtStartAndEnd(line)
        IF line does not start with '>' THEN
            taille <- taille + Length(line)
        END IF
    END FOR
CLOSE file
Return taille









Method name ( ): parse_scanformatches
parse_scanformatches(input_file ↓ )

Description :
Parses a motif search results file and extracts relevant information.

Header:
input_file ↓ : path to the motif search results file

Return value:
- tuple (seq_id: String, positions: Dictionary)

Body
Local variables
- seq_id: String
- positions: Dictionary
- i: Integer
- line: String
- seq_id_match: String
- pos_match: String
- start_match : String

Instructions

seq_id <- Undefined
positions <- empty Dictionary
i <- 0
OPEN input_file FOR READING DO
    FOR EACH line IN file DO
        line <- stripSpacesAtStartAndEnd(line)
        IF line starts with '>' and seq_id = Undefined THEN
            seq_id <- ExtractWithRegex('^>(\\S+):' , line)
        END IF
        IF line starts with '>' THEN
            i <- i + 1
            pos_match <- ExtractWithRegex(':\\[(\\d*),(\\d*)]' ,line)
            positions[i] <- pos_match
        END IF
        start_match <- ExtractWithRegex('\\b(atg|ttg|gtg)\\b' , line)
        IF start_match is not empty THEN
            Add start_match to positions[i]
        END IF
    END FOR
CLOSE file

Return seq_id, positions






Method name ( ): Write_GFF3
Write_GFF3(input_file ↓ , output_file ↓ , feature_type ↓ ,  seq_fasta_file (Undefined by default))

Description :
Write a GFF3 file from scan_for_matches output.
It uses information obtained from parse_scanformatches.
Header:
input_file ↓ : path to the motif search results file
output_file ↓ : path to the output GFF3 file
feature_type ↓ : feature type to annotate in the GFF3 file
seq_fasta_file ↓ : path to the FASTA file (optional)

Return value:
Confirmation message for writing the GFF3 file.

Body
Local variables
- seq_id: String
- positions: Dictionary
- source : String
- taille : Integer
- feature_type : String
- codon_start : String
- start : String
- end : String
- gff_start : String
- gff_end : String
- score : String
- strand : String
- phase : String
- attributes : String
- gff3_line : String

Instructions

seq_id, positions <- parse_scanformatches(input_file)
source <- "scan_for_matches"
taille <- Undefined
IF seq_fasta_file is not Undefined THEN
    taille <- taille_seq(seq_fasta_file)
END IF
OPEN output_file FOR WRITING DO
    WriteLine <- ( "##gff-version 3")
    Write WriteLine TO output_file

    If taille is defined THEN
        WriteLine <- ('##sequence-region' +  seq_id + '1' + taille + '\n')
        Write WriteLine TO output_file
    END IF
    FOR EACH i IN Sort(positions) DO
        codon_start <- Undefined
        If Sizes(positions[i]) = 3 THEN
            start , end ,codon_start <- positions[i]
        ELSE
            start , end <- positions[i]
        END IF
        gff_start <- Minimum( Integer(start) , Integer(end) )
        gff_end <- Maximum( Integer(start) , Integer(end) )
        score <- "."
        strand <- IF Integer(start) < Integer(end) THEN '+' ELSE '-'
        phase <- "." // no phase for RBS/promoter/terminator

        IF codon_start is not Undefined THEN
            attributes <- "ID=" + feature_type + "_" + i + ";Note=" + source + "_prediction;Start_codon=" + codon_start
        ELSE
            attributes <- "ID=" + feature_type + "_" + i + ";Note=" + source + "_prediction"
        END IF
        WriteLine <- ( seq_id + '\t' + source + '\t' + feature_type + '\t' + gff_start + '\t' + gff_end + '\t' + score + '\t' + strand + '\t' + phase + '\t' + attributes )
        Write WriteLine TO output_file
    END FOR
CLOSE output_file
Return "GFF3 file '" + output_file + "' written successfully."
