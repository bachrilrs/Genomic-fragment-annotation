Method name ( ): extract_cds_genemark
extract_cds_genemark(input_file ↓)
    Description:
    // Extracts CDS from a GeneMark output file and returns a list of “unique” CDS (one CDS per group sharing the same strand, frame, and stop codon).

    Header:
    input_file : path to the GeneMark input file

    Body

    Local variables
        • cds_list : List
        • state : String
        • line : String
        • fields : List
        • groups : Dictionary<(strand, frame, stop), CDS>
        • fields : list
        • strand : string (‘+’ or ‘–’)
        • frame : Integer
        • stop : Integer
        • key : Tuple (strand, frame, stop)
        
    Returns:
    List of unique CDS extracted from the GeneMark file
Instructions

cds_list <- empty list 
state <- 'outside'

OPEN input_file FOR READING DO
    FOR EACH line IN input_file DO
    line <- stripSpacesAtStartAndEnd(line)

    If state = 'outside' THEN
        If line contains 'List of Open reading frames'
        state <- 'header'
        END IF
        continue
    END IF
    IF state = 'header' THEN
        IF line starts with "--------" THEN
            state <- "data"
        END IF
            continue
    END IF
        
    If line contains "List of Regions of interest" or line contains "ABOUT THE MATRIX USED" THEN
        EXIT THE FOR LOOP
    END IF
    
    IF line is empty OR line starts with "Left" OR line starts with "end" OR line starts with "--------" THEN
        continue
    END IF
        
    IF FirstLetter(line) is a digit AND (line contains "direct" OR line contains "complement") AND (line contains "fr") THEN
        fields <- SplitBySpaces(line) 
        cds.left <- Integer(fields[0])
        cds.right <- Integer(fields[1])
        cds.orientation <- fields[2]  
        cds.frame <- Integer(fields[4])
        cds.coding_prob <- Real(fields[5])
        cds.start_prob <- Real(fields[6])
        Add cds to cds_list
    END IF
    END FOR

CLOSE input_file
Return cds_list // list of CDS  
End Instructions
End Function



Method name ( ): extract_info_Genemark

Description:
// Extracts the sequence identifier, source program name, and sequence length from a GeneMark output file.

Header:
input_file ↓ : path to the GeneMark input file
Return value: Tuple containing:
    sequence identifier (seqid)
    program name (source)
    sequence length (taille)

Body
    Local variables
    - seqid_pat : String
    - taille_pat : String
    - id : String
    - source : String
    - taille : Integer
    - line : String

Instructions

seqid_pat <- 'Sequence:\s+(\S+)'
taille_pat <- 'Sequence length:\s+(\d+)'
id <- empty string
source <- 'unknown'
taille <- '.'

OPEN input_file FOR READING DO 
    FOR EACH line IN input_file DO
    
    IF line starts with 'Sequence:' THEN
        seq <- ExtractWithRegex(seqid_pat, line)
        If ExtractWithRegex returns a value THEN
        id <- extracted value
        ELSE
            id <- 'unknown'
    END IF
    If line contains 'GeneMark' THEN
        source <- 'GeneMark'
    END IF
    IF line starts with 'Sequence length:' THEN
            taille <- ExtractWithRegex(taille_pat, line)
        IF ExtractWithRegex returns a value THEN
            taille <- Integer(extracted value)
        END IF
    END IF

    IF id is not '' and source is not 'unknown' AND taille is not '.' THEN
        EXIT THE LOOP
    END IF
    END FOR
CLOSE input_file
End Instructions
Return (id, source, taille)





Method name ( ): Write GFF3_GeneMark
Write GFF3_GeneMark(input_file ↓ , output_file ↓ )
Description:
// Writes GFF3 annotations of CDS extracted from a GeneMark output file
// This function uses ExtractInfos_GeneMark and ExtractCDS_GeneMark to obtain the required information.

Header:
    input_file ↓ : path to the GeneMark input file
    output_file ↓ : path to the GFF3 output file
    Returns success message

Body
    Local variables
    - seqid : String
    - source : String
    - taille : Integer
    - cds_list : List
    - feature_type : String
    - start : Integer
    - end : Integer
    - i : Integer
    - strand : String
    - fields : List
    - phase : Integer
    - score : Real

OPEN output_file FOR WRITING DO 
    (seqid, source, taille) <- ExtractInfos_GeneMark(input_file)
    cds_list <- ExtractCDS_GeneMark(input_file)
    WRITE "##gff-version 3\n" TO output_file
    WRITE "##sequence-region " + seqid + " 1 " + taille + "\n" TO output_file

FOR i from 0 to length(cds_list) − 1 DO 
    fields <- cds_list[i]
    start <- Minimum(Integer(fields[0]), Integer(fields[1]))
    end <- Maximum(Integer(fields[0]), Integer(fields[1]))
    feature_type <- 'CDS'
    IF fields[2] = 'direct' THEN 
      strand <- '+' 
    ELSE 
      strand <- '-'
    END IF
    phase <- fields[4] − 1 // adjust frame for GFF3 (0-based)

    score <- fields[5] IF SCORE exists, otherwise "."
    END IF

    atributes  <- 'ID=cds' + (i + 1) + ';Note' + (source) + '_prediction'

    ecrire_ligne <- seqid + '\t' + source + '\t' + feature_type + '\t' + start + '\t' + end + '\t' + score + '\t' + strand + '\t' + phase + '\t' + atributes + '\n'
    WRITE ecrire_ligne TO output_file
END FOR
CLOSE output_file
Return 'GFF3 file ' + output_file + 'written successfully'
